初回セットアップが必要です

Supabaseの SQL Editor で以下を1回実行してから「再読み込み」を押してください。（テーブルと写真保存用バケットを作成します）

-- 点検データ本体
create table if not exists kankyo_inspections (
  id uuid primary key default gen_random_uuid(),
  cycle text not null,
  inspection_date date,
  department text not null,
  sort_order int default 0,
  items jsonb not null default '[]',
  homework text default '',
  prev_homework text default '',
  updated_at timestamptz default now(),
  unique (cycle, department)
);
alter table kankyo_inspections enable row level security;
create policy "kankyo auth all" on kankyo_inspections
  for all to authenticated using (true) with check (true);

-- 写真保存用バケット（公開）
insert into storage.buckets (id, name, public)
values ('kankyo-photos','kankyo-photos', true)
on conflict (id) do nothing;

create policy "kankyo photo insert" on storage.objects
  for insert to authenticated with check (bucket_id = 'kankyo-photos');
create policy "kankyo photo select" on storage.objects
  for select using (bucket_id = 'kankyo-photos');
create policy "kankyo photo delete" on storage.objects
  for delete to authenticated using (bucket_id = 'kankyo-photos');
